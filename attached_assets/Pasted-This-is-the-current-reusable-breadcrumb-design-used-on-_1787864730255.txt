This is the current reusable breadcrumb design used on mobile:

```tsx
import Link from "next/link";
import type { Crumb } from "@/lib/schema";
import { breadcrumbItems } from "@/lib/schema";

export default function Breadcrumbs({ crumbs }: { crumbs: Crumb[] }) {
  const items = breadcrumbItems(crumbs);

  return (
    <nav
      aria-label="Navigačná cesta"
      className="
        border-b border-gray-100 bg-white
        lg:absolute lg:left-0 lg:right-0 lg:top-28 lg:z-30
        lg:border-0 lg:bg-transparent
      "
    >
      <div
        className="
          container mx-auto max-w-7xl
          px-4 py-3 sm:px-6
          lg:px-8 lg:py-1
        "
      >
        <ol
          className="
            flex min-w-0 items-center gap-2
            overflow-x-auto whitespace-nowrap
            text-xs font-semibold text-gray-500
            [scrollbar-width:none]
            [&::-webkit-scrollbar]:hidden

            lg:w-fit lg:max-w-full
            lg:rounded-full
            lg:bg-white/90
            lg:px-3 lg:py-2
            lg:shadow-sm
            lg:ring-1 lg:ring-black/5
            lg:backdrop-blur-sm
          "
        >
          {items.map((item, index) => {
            const isCurrent = index === items.length - 1;

            return (
              <li
                key={`${item.path}-${item.name}`}
                className="
                  flex min-w-0 shrink-0
                  items-center gap-2
                "
              >
                {index > 0 && (
                  <svg
                    aria-hidden="true"
                    className="h-3.5 w-3.5 shrink-0 text-gray-300"
                    viewBox="0 0 24 24"
                    fill="none"
                    stroke="currentColor"
                    strokeWidth={2.5}
                  >
                    <path d="m9 5 7 7-7 7" />
                  </svg>
                )}

                {isCurrent ? (
                  <span
                    aria-current="page"
                    className="
                      max-w-[min(18rem,55vw)]
                      truncate text-midnight
                    "
                  >
                    {item.name}
                  </span>
                ) : (
                  <Link
                    href={item.path}
                    className="
                      shrink-0
                      transition-colors
                      hover:text-accent
                      focus-visible:outline-none
                      focus-visible:ring-2
                      focus-visible:ring-accent
                      focus-visible:ring-offset-2
                    "
                  >
                    {item.name}
                  </Link>
                )}
              </li>
            );
          })}
        </ol>
      </div>
    </nav>
  );
}
```

The mobile behavior comes mainly from:

```txt
flex
overflow-x-auto
whitespace-nowrap
gap-2
text-xs
py-3
[scrollbar-width:none]
[&::-webkit-scrollbar]:hidden
```

This keeps the breadcrumb on one line, allows horizontal scrolling on narrow screens, hides the scrollbar, and truncates only the current page label when it is too long.
