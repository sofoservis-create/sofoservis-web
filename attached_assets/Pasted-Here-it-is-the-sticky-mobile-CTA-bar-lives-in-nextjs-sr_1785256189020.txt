Here it is — the sticky mobile CTA bar lives in nextjs/src/components/StickyMobileCta.tsx (mounted globally in nextjs/src/app/layout.tsx):

"use client";
import { useEffect, useState } from "react";
import { BUSINESS } from "@/lib/business";
/**
 * Persistent bottom CTA bar on mobile. Appears after the visitor scrolls past
 * the hero so it never competes with the primary above-the-fold form.
 */
export default function StickyMobileCta() {
  const [visible, setVisible] = useState(false);
  useEffect(() => {
    const onScroll = () => setVisible(window.scrollY > 600);
    onScroll();
    window.addEventListener("scroll", onScroll, { passive: true });
    return () => window.removeEventListener("scroll", onScroll);
  }, []);
  return (
    <div
      aria-hidden={!visible}
      className={`lg:hidden fixed bottom-0 inset-x-0 z-40 transition-transform duration-300 ${
        visible ? "translate-y-0" : "translate-y-full pointer-events-none"
      }`}
    >
      <div className="bg-white/95 backdrop-blur border-t border-gray-200 shadow-[0_-4px_16px_rgba(0,0,0,0.08)] px-3 py-2.5 pb-[calc(0.625rem+env(safe-area-inset-bottom))] flex gap-2">
        <a
          href="/kontakt"
          className="flex-1 min-h-[48px] inline-flex items-center justify-center gap-2 bg-accent text-white font-black rounded-xl text-sm"
        >
          Cenová ponuka
          <svg className="w-4 h-4" viewBox="0 0 24 24" fill="none" stroke="currentColor" strokeWidth={2.5}>
            <path d="M5 12h14M12 5l7 7-7 7" />
          </svg>
        </a>
        <a
          href={`tel:${BUSINESS.phone}`}
          className="flex-1 min-h-[48px] inline-flex items-center justify-center gap-2 bg-white border-2 border-accent text-accent font-black rounded-xl text-sm"
        >
          <svg className="w-4 h-4" fill="none" viewBox="0 0 24 24" stroke="currentColor" strokeWidth={2}>
            <path d="M22 16.92v3a2 2 0 0 1-2.18 2 19.79 19.79 0 0 1-8.63-3.07 19.5 19.5 0 0 1-6-6 19.79 19.79 0 0 1-3.07-8.67A2 2 0 0 1 4.11 2h3a2 2 0 0 1 2 1.72 12.84 12.84 0 0 0 .7 2.81 2 2 0 0 1-.45 2.11L8.09 9.91a16 16 0 0 0 6 6l1.27-1.27a2 2 0 0 1 2.11-.45 12.84 12.84 0 0 0 2.81.7A2 2 0 0 1 22 16.92z" />
          </svg>
          Zavolať
        </a>
      </div>
    </div>
  );
}

Key behavior:

Mobile-only (lg:hidden), fixed to the bottom, above content (z-40).
Appears after 600px of scroll — slides up/down with a 300ms transform transition so it doesn't compete with the hero form.
"Cenová ponuka" → filled accent button linking to /kontakt; "Zavolať" → outlined button with tel: link using the phone from @/lib/business.
Respects the iPhone home-indicator area via env(safe-area-inset-bottom)
